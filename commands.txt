// 30-seconds-of-java8 (java 8)
mvn clean compile test-compile dependency:copy-dependencies \
  -f "$(pwd)/samples/30-seconds-of-java8/pom.xml" \
  -DskipTests -DoutputDirectory=lib \
  -DincludeScope=test && \
java -cp target/astor-*-jar-with-dependencies.jar fr.inria.main.evolution.AstorMain \
  -mode custom \
  -customengine fr.inria.astor.approaches.flakyrepair.FrEngine \
  -javacompliancelevel 8 \
  -srcjavafolder src/java/ \
  -srctestfolder src/test/  \
  -binjavafolder target/classes/ \
  -bintestfolder target/test-classes/ \
  -location "$(pwd)/samples/30-seconds-of-java8/" \
  -dependencies "$(pwd)/samples/30-seconds-of-java8/lib"

// Docker
// Build
docker build \
  --build-arg USER_UID=$(id -u) \
  --build-arg USER_GID=$(id -g) \
  -t astor .

// Rodar
PROJECT_NAME="array" && \
PROJECT_PATH="$(pwd)/minimal_samples/array" && \
JAVA_VERSION=11 && \
RESULT_PATH="$(pwd)/results/$PROJECT_NAME/latest" && \
mkdir -p "$RESULT_PATH" && \
docker container inspect astor-docker > /dev/null 2>&1 || \
  docker create --name astor-docker \
    -v "$RESULT_PATH:/astor/output_astor" \
    -e JAVA_VERSION=$JAVA_VERSION \
    -e ENGINE_CLASS=fr.inria.astor.approaches.flakyseeding.FsEngine \
    astor && \
docker exec -it astor-docker rm -rf astor-docker/target-project && \
docker cp "$PROJECT_PATH/." astor-docker/target-project" && \
docker start -a astor-docker


JAVA_VERSION = Versão do java usado.
ENGINE_CLASS = Método do ASTOR a ser usado.
