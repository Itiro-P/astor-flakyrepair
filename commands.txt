// Compilar o projeto

mvn clean package -DskipTests=true

// Compilar um projeto
mvn clean test-compile -f "$(pwd)/samples/30-seconds-of-java8/pom.xml"

// Copiar as dependências
mvn dependency:copy-dependencies -f "$(pwd)/samples/30-seconds-of-java8/pom.xml" -DoutputDirectory=lib -DincludeScope=test

// Compilar e executar com um projeto

// 30-seconds-of-java8 (java 8)
mvn clean compile test-compile dependency:copy-dependencies \
  -f "$(pwd)/samples/30-seconds-of-java8/pom.xml" \
  -DskipTests=true -DoutputDirectory=lib \
  -DincludeScope=test && \
java -cp target/astor-*-jar-with-dependencies.jar fr.inria.main.evolution.AstorMain \
  -mode custom \
  -customengine fr.inria.astor.approaches.flakyrepair.FlakyRepairEngine \
  -javacompliancelevel 8 \
  -srcjavafolder src/java/ \
  -srctestfolder src/test/  \
  -binjavafolder target/classes/ \
  -bintestfolder target/test-classes/ \
  -location "$(pwd)/samples/30-seconds-of-java8/" \
  -dependencies "$(pwd)/samples/30-seconds-of-java8/lib"
